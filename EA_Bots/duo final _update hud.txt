double GetVal(int handle, int buffer=0, int index=0) {
   double val[];
   ArraySetAsSeries(val, true);
   if(CopyBuffer(handle, buffer, index, 1, val) > 0) return val[0];
   return 0;
}

void OnTick() {
   if(IsNFPPauseActive() || IsFridayExit()) { 
      BotActive = false; 
      tradeReason = IsFridayExit() ? "FRIDAY EXIT" : "NFP PAUSE";
      return; 
   }

   // Fetch Values
   double f0 = GetVal(hFast);
   double s0 = GetVal(hSlow);
   double i0 = GetVal(hIntra);
   double t0 = GetVal(hTrend);
   double d1 = GetVal(hD1);
   double adx0 = GetVal(hADX);
   double mom0 = GetVal(hMom);
   double atrVal = GetVal(hATR);
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);

   bool isBullTrend = (i0 > t0 && bid > d1);
   bool isBearTrend = (i0 < t0 && bid < d1);

   // Pullback Logic
   if(isBullTrend && !pullbackActive && bid < f0) {
      pullbackActive = true; lastPullbackType = "BULLISH";
   } else if(pullbackActive && lastPullbackType == "BULLISH" && bid > f0) {
      pullbackActive = false;
      if(GetActiveCount() < MaxTrades) Execute(ORDER_TYPE_BUY, atrVal, 1.0);
   }

   if(isBearTrend && !pullbackActive && ask > f0) {
      pullbackActive = true; lastPullbackType = "BEARISH";
   } else if(pullbackActive && lastPullbackType == "BEARISH" && ask < f0) {
      pullbackActive = false;
      if(GetActiveCount() < MaxTrades) Execute(ORDER_TYPE_SELL, atrVal, 1.0);
   }

   if(EnableTrendRide) ManageTrendRide(TrendRideATR, TrendRideEMA);
   
   DrawTerminalHUD(adx0, mom0, atrVal, 0, "N/A", i0);
}

bool Execute(ENUM_ORDER_TYPE type, double atr, double mult) {
    double lot = (ManualLotSize > 0) ? ManualLotSize : 0.01;
    double sl = (type == ORDER_TYPE_BUY) ? 
                SymbolInfoDouble(_Symbol, SYMBOL_BID) - (atr * ATR_SL_Mult) : 
                SymbolInfoDouble(_Symbol, SYMBOL_ASK) + (atr * ATR_SL_Mult);

    if(type == ORDER_TYPE_BUY) return trade.Buy(lot, _Symbol, 0, sl, 0);
    if(type == ORDER_TYPE_SELL) return trade.Sell(lot, _Symbol, 0, sl, 0);
    return false;
}

int GetActiveCount() {
   int count = 0;
   for(int i=PositionsTotal()-1; i>=0; i--) {
      if(posInfo.SelectByIndex(i) && posInfo.Symbol() == _Symbol && posInfo.Magic() == 123456) count++;
   }
   return count;
}

void ManageTrendRide(double atrMultiplier, int emaPeriod) {
    double atr = GetVal(hATR);
    double ema = GetVal(hIntra); // Using IntraEMA as proxy for ride

    for(int i=PositionsTotal()-1; i>=0; i--) {
        if(posInfo.SelectByIndex(i) && posInfo.Magic() == 123456) {
            double currentSL = posInfo.StopLoss();
            double newSL = (posInfo.PositionType() == POSITION_TYPE_BUY) ? 
                            ema - (atr * atrMultiplier) : ema + (atr * atrMultiplier);

            if(posInfo.PositionType() == POSITION_TYPE_BUY && newSL > currentSL)
                trade.PositionModify(posInfo.Ticket(), newSL, 0);
            if(posInfo.PositionType() == POSITION_TYPE_SELL && (newSL < currentSL || currentSL == 0))
                trade.PositionModify(posInfo.Ticket(), newSL, 0);
        }
    }
}

bool IsFridayExit() {
   MqlDateTime dt;
   TimeCurrent(dt);
   return (dt.day_of_week == 5 && dt.hour >= ExitHourFriday);
}

bool IsNFPPauseActive() {
   // Simplified logic: Check if it's the first Friday of the month
   MqlDateTime dt;
   TimeCurrent(dt);
   return (dt.day_of_week == 5 && dt.day <= 7 && dt.hour >= 12 && dt.hour <= 16);
}
