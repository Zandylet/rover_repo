//+------------------------------------------------------------------+
//| KIT-ARA Trend Rider (Error-Free MT5 EA)                          |
//| Trend Continuation + Pullback Entries + Scale-In                 |
//+------------------------------------------------------------------+
#property strict
#property version "3.20"

#include <Trade/Trade.mqh>
#include <Trade/PositionInfo.mqh>

CTrade trade;
CPositionInfo pos;

//================ INPUTS =================//
input double LotSize        = 0.01;
input int    MaxTrades      = 3;
input int    Magic          = 123456;

input int FastEMA  = 9;
input int SlowEMA  = 21;
input int IntraEMA = 55;
input int TrendEMA = 200;

input int ADXPeriod = 14;
input int ATRPeriod = 14;
input int MomPeriod = 14;
input int MinADX    = 22;

input double PullbackATRMin    = 0.3;
input double PullbackATRMax    = 1.0;
input double LargePullbackATR = 1.5;

input double BreakEvenRR = 1.0;
input double LockRR      = 4.0;
input double TrailATR    = 1.0;

input bool EnableScaleIn = true;

//================ HANDLES =================//
int hFast,hSlow,hIntra,hTrend,hHTF,hD1,hADX,hATR,hMom;

//================ INIT =================//
int OnInit()
{
   hFast  = iMA(_Symbol,PERIOD_M15,FastEMA,0,MODE_EMA,PRICE_CLOSE);
   hSlow  = iMA(_Symbol,PERIOD_M15,SlowEMA,0,MODE_EMA,PRICE_CLOSE);
   hIntra = iMA(_Symbol,PERIOD_M15,IntraEMA,0,MODE_EMA,PRICE_CLOSE);
   hTrend = iMA(_Symbol,PERIOD_M15,TrendEMA,0,MODE_EMA,PRICE_CLOSE);

   hHTF   = iMA(_Symbol,PERIOD_H1,200,0,MODE_EMA,PRICE_CLOSE);
   hD1    = iMA(_Symbol,PERIOD_D1,200,0,MODE_EMA,PRICE_CLOSE);

   hADX   = iADX(_Symbol,PERIOD_M15,ADXPeriod);
   hATR   = iATR(_Symbol,PERIOD_M15,ATRPeriod);
   hMom   = iMomentum(_Symbol,PERIOD_M15,MomPeriod,PRICE_CLOSE);

   trade.SetExpertMagicNumber(Magic);
   return INIT_SUCCEEDED;
}

//================ UTILITIES =================//
int ActiveTrades()
{
   int c=0;
   for(int i=PositionsTotal()-1;i>=0;i--)
      if(pos.SelectByIndex(i) && pos.Magic()==Magic)
         c++;
   return c;
}

double GetRR(ulong ticket)
{
   if(!pos.Select(ticket)) return 0.0;

   double entry = pos.PriceOpen();
   double sl    = pos.StopLoss();
   if(sl<=0) return 0.0;

   double price = (pos.PositionType()==POSITION_TYPE_BUY)
                  ? SymbolInfoDouble(_Symbol,SYMBOL_BID)
                  : SymbolInfoDouble(_Symbol,SYMBOL_ASK);

   return MathAbs(price-entry)/MathAbs(entry-sl);
}

//================ PULLBACK LOGIC =================//
bool ValidPullback(bool buy,double price,double ema,double atr,double mom)
{
   double depth = MathAbs(price-ema);

   bool depthOK = depth >= atr*PullbackATRMin &&
                  depth <= atr*PullbackATRMax;

   bool momOK  = buy ? mom>100.0 : mom<100.0;
   bool sideOK = buy ? price>ema : price<ema;

   return depthOK && momOK && sideOK;
}

bool LargePullback(bool buy,double price,double ema,double atr,double mom)
{
   bool emaBreak = buy ? price<ema : price>ema;
   bool momFlip  = buy ? mom<100.0 : mom>100.0;
   bool deep     = MathAbs(price-ema) >= atr*LargePullbackATR;

   return emaBreak && momFlip && deep;
}

//================ POSITION MANAGEMENT =================//
void ManagePositions(double atr,double ema,double mom)
{
   for(int i=PositionsTotal()-1;i>=0;i--)
   {
      if(!pos.SelectByIndex(i) || pos.Magic()!=Magic) continue;

      ulong  ticket = pos.Ticket();
      bool   buy    = pos.PositionType()==POSITION_TYPE_BUY;
      double entry  = pos.PriceOpen();
      double sl     = pos.StopLoss();
      double price  = buy ? SymbolInfoDouble(_Symbol,SYMBOL_BID)
                          : SymbolInfoDouble(_Symbol,SYMBOL_ASK);

      double rr = GetRR(ticket);

      // Break-even
      if(rr>=BreakEvenRR && sl<=0)
      {
         trade.PositionModify(ticket,entry,0);
      }

      // Lock profits
      if(rr>=LockRR)
      {
         double lockSL = buy ? entry+atr : entry-atr;
         if((buy && lockSL>sl) || (!buy && lockSL<sl))
            trade.PositionModify(ticket,lockSL,0);
      }

      // Trailing stop
      double trailSL = buy ? price-atr*TrailATR : price+atr*TrailATR;
      if((buy && trailSL>sl) || (!buy && trailSL<sl))
         trade.PositionModify(ticket,trailSL,0);

      // Exit on large pullback
      if(LargePullback(buy,price,ema,atr,mom))
      {
         trade.PositionClose(ticket);
         PrintFormat("EXIT | Large Pullback | RR: %.2f", rr);
      }
   }
}

//================ MAIN =================//
void OnTick()
{
   double f[],s[],i[],t[],htf[],d1[],adx[],atr[],mom[];
   ArraySetAsSeries(f,true);ArraySetAsSeries(s,true);
   ArraySetAsSeries(i,true);ArraySetAsSeries(t,true);
   ArraySetAsSeries(htf,true);ArraySetAsSeries(d1,true);
   ArraySetAsSeries(adx,true);ArraySetAsSeries(atr,true);
   ArraySetAsSeries(mom,true);

   if(CopyBuffer(hFast,0,0,1,f)<1) return;
   if(CopyBuffer(hSlow,0,0,1,s)<1) return;
   if(CopyBuffer(hIntra,0,0,1,i)<1) return;
   if(CopyBuffer(hTrend,0,0,1,t)<1) return;
   if(CopyBuffer(hHTF,0,0,1,htf)<1) return;
   if(CopyBuffer(hD1,0,0,1,d1)<1) return;
   if(CopyBuffer(hADX,0,0,1,adx)<1) return;
   if(CopyBuffer(hATR,0,0,1,atr)<1) return;
   if(CopyBuffer(hMom,0,0,1,mom)<1) return;

   if(adx[0]<MinADX) return;

   double bid = SymbolInfoDouble(_Symbol,SYMBOL_BID);
   double ask = SymbolInfoDouble(_Symbol,SYMBOL_ASK);

   bool bull = i[0]>t[0] && bid>d1[0] && bid>htf[0];
   bool bear = i[0]<t[0] && bid<d1[0] && bid<htf[0];

   bool buyBreak  = bull && f[0]>i[0] && s[0]>i[0] && mom[0]>100.0;
   bool sellBreak = bear && f[0]<i[0] && s[0]<i[0] && mom[0]<100.0;

   // Initial entries
   if(ActiveTrades()<MaxTrades)
   {
      if(buyBreak)  trade.Buy(LotSize,_Symbol);
      if(sellBreak) trade.Sell(LotSize,_Symbol);
   }

   // Scale-in logic
   if(EnableScaleIn && ActiveTrades()<MaxTrades)
   {
      for(int p=PositionsTotal()-1;p>=0;p--)
      {
         if(!pos.SelectByIndex(p) || pos.Magic()!=Magic) continue;
         if(GetRR(pos.Ticket())<1.0) continue;

         bool buy = pos.PositionType()==POSITION_TYPE_BUY;
         double price = buy?bid:ask;

         if(ValidPullback(buy,price,i[0],atr[0],mom[0]))
         {
            if(buy) trade.Buy(LotSize,_Symbol);
            else    trade.Sell(LotSize,_Symbol);

            Print("SCALE-IN | Valid Trend Pullback");
            break;
         }
      }
   }

   ManagePositions(atr[0],i[0],mom[0]);
}
